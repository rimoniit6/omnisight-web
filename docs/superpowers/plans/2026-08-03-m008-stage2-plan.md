## Task 1: Analytics Cache Layer

**Files:**
- Create: `src/lib/analytics/cache.ts`
- Modify: `src/lib/analytics/index.ts` (add export)

**Interfaces:**
- Consumes: nothing external (in-memory only)
- Produces: `AnalyticsCache` class with `get`/`set`/`del`/`invalidate`/`invalidateTag`/`clear`

**Design:**
- TTL-based entry expiration (default 30s configurable)
- Cache tags for grouping (e.g., `dashboard:7d`, `rollups:today`)
- Background refresh (stale-while-revalidate pattern)
- Memory-safe: max entries count, LRU eviction
- Cache key normalization (stable string from params)
- `invalidateAfterRebuild()` to clear all cache after a rebuild

- [ ] **Step 1: Write the cache module**

```typescript
import { LRUCache } from '@/lib/lru-cache'

export interface CacheEntry<T> {
  value: T
  expiresAt: number
  tags: string[]
  lastAccess: number
}

export class AnalyticsCache {
  private store = new Map<string, CacheEntry<unknown>>()
  private tagIndex = new Map<string, Set<string>>()
  private readonly maxEntries: number
  private readonly defaultTtlMs: number

  constructor(opts: { maxEntries?: number; defaultTtlMs?: number } = {}) {
    this.maxEntries = opts.maxEntries ?? 500
    this.defaultTtlMs = opts.defaultTtlMs ?? 30_000
  }

  key(parts: unknown[]): string {
    return parts.map((p) => JSON.stringify(p)).join(':')
  }

  get<T>(k: string): T | undefined {
    const entry = this.store.get(k)
    if (!entry) return undefined
    if (Date.now() > entry.expiresAt) {
      this.store.delete(k)
      entry.tags.forEach((t) => this.tagIndex.get(t)?.delete(k))
      return undefined
    }
    entry.lastAccess = Date.now()
    return entry.value as T
  }

  set<T>(k: string, value: T, ttlMs?: number, tags: string[] = []): void {
    this.evictIfNeeded()
    this.store.set(k, { value, expiresAt: Date.now() + (ttlMs ?? this.defaultTtlMs), tags, lastAccess: Date.now() })
    for (const tag of tags) {
      const set = this.tagIndex.get(tag) ?? new Set()
      set.add(k)
      this.tagIndex.set(tag, set)
    }
  }

  del(k: string): void {
    const entry = this.store.get(k)
    if (entry) {
      entry.tags.forEach((t) => this.tagIndex.get(t)?.delete(k))
    }
    this.store.delete(k)
  }

  invalidateTag(tag: string): number {
    const set = this.tagIndex.get(tag)
    if (!set) return 0
    let count = 0
    for (const k of set) {
      this.store.delete(k)
      count++
    }
    this.tagIndex.delete(tag)
    return count
  }

  clear(): number {
    const n = this.store.size
    this.store.clear()
    this.tagIndex.clear()
    return n
  }

  stats(): { entries: number; memoryEstimateBytes: number } {
    return { entries: this.store.size, memoryEstimateBytes: this.store.size * 512 }
  }

  private evictIfNeeded(): void {
    if (this.store.size < this.maxEntries) return
    // LRU: remove oldest by lastAccess
    const sorted = [...this.store.entries()].sort((a, b) => a[1].lastAccess - b[1].lastAccess)
    const toEvict = sorted.slice(0, Math.floor(this.maxEntries * 0.1))
    for (const [k] of toEvict) {
      this.del(k)
    }
  }

  /** Clear all analytics cache after a rebuild — no stale analytics. */
  invalidateAfterRebuild(): number {
    return this.clear()
  }
}

export const analyticsCache = new AnalyticsCache()
```

```typescript
import { db } from '@/lib/db'

export type AnalyticsJobType = 'incremental_rollup' | 'rebuild' | 'stale_repair' | 'retention_cleanup' | 'daily_summary' | 'alert_evaluation'

export interface AnalyticsQueueJob {
  id: string
  jobType: AnalyticsJobType
  mode: string
  status: 'pending' | 'running' | 'completed' | 'failed'
  retryCount: number
  maxRetries: number
  lastError: string | null
  startedAt: Date | null
  finishedAt: Date | null
  durationMs: number | null
  nextRunAt: Date | null
  payload: Record<string, unknown> | null
  createdAt: Date
  updatedAt: Date
}

const INITIAL_DELAYS_MS = [0, 5_000, 15_000, 60_000, 300_000] // exponential: 0, 5s, 15s, 1m, 5m

export async function enqueueAnalyticsJob(opts: {
  jobType: AnalyticsJobType
  mode?: string
  payload?: Record<string, unknown>
  runAt?: Date
}): Promise<string> {
  const job = await db.AnalyticsJob.create({
    data: {
      jobType: opts.jobType,
      mode: opts.mode ?? 'incremental',
      status: 'pending',
      fromDate: opts.runAt ?? null,
      toDate: opts.runAt ?? null,
    },
  })
  return job.id
}

export async function dequeueDueJobs(limit = 10): Promise<AnalyticsQueueJob[]> {
  const jobs = await db.analyticsJob.findMany({
    where: { status: 'running' }, // stale — detect jobs stuck in 'running'
  })
  // Mark stale jobs back to pending
  for (const j of jobs) {
    if (j.startedAt && Date.now() - j.startedAt.getTime() > 300_000) {
      await db.analyticsJob.update({ where: { id: j.id }, data: { status: 'pending', lastError: 'stale' } })
    }
  }
  
  const due = await db.analyticsJob.findMany({
    where: { status: 'pending', OR: [{ fromDate: null }, { fromDate: { lte: new Date() } }] },
    orderBy: { createdAt: 'asc' },
    take: limit,
  })
  return due
}

export async function claimJob(jobId: string): Promise<void> {
  await db.analyticsJob.update({
    where: { id: jobId },
    data: { status: 'running', startedAt: new Date() },
  })
}

export async function markJobComplete(jobId: string, rows: number): Promise<void> {
  const job = await db.analyticsJob.findUnique({ where: { id: jobId } })
  const duration = job?.startedAt ? Date.now() - job.startedAt.getTime() : null
  await db.analyticsJob.update({
    where: { id: jobId },
    data: {
      status: 'completed',
      rowsProcessed: rows,
      finishedAt: new Date(),
      durationMs,
    },
  })
}

export async function markJobFailed(jobId: string, error: string): Promise<void> {
  const job = await db.analyticsJob.findUnique({ where: { id: jobId } })
  const duration = job?.startedAt ? Date.now() - job.startedAt.getTime() : null
  const retryCount = (job?.error ? 1 : 0) + 1
  const delay = INITIAL_DELAYS_MS[Math.min(retryCount, INITIAL_DELAYS_MS.length - 1)]
  await db.analyticsJob.update({
    where: { id: jobId },
    data: {
      status: 'pending',
      error,
      rowsProcessed: retryCount,
      finishedAt: new Date(),
      durationMs: duration,
      fromDate: delay ? new Date(Date.now() + delay) : null, // nextRunAt via fromDate
    },
  })
}

export async function getJobHistory(opts: { limit?: number; status?: string } = {}): Promise<AnalyticsQueueJob[]> {
  const where: Record<string, unknown> = {}
  if (opts.status) where.status = opts.status
  const jobs = await db.analyticsJob.findMany({
    where,
    orderBy: { createdAt: 'desc' },
    take: opts.limit ?? 100,
  })
  return jobs
}
```
